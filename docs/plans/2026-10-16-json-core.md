## Global Constraints

- `yt::json` uses the standard library only and compiles as C++17 (`tst_json` is built with `STD17` to prove it). No new third-party dependency in the library; `bench/` fetches its comparison libraries into an ignored `bench/ext/`.
- Documents: nesting ≤ `kMaxDepth` = 128, size < `kMaxSize` = 512 MiB; anything malformed leaves `Doc::ok()` false and every lookup absent. No exceptions, no recursion over input (iterative walks only).
- UTF-8 is not validated (bytes pass through; a lone `\uD800` decodes as WTF-8) — the spec's documented choice.
- Style (AGENTS.md §5–§6): 4 spaces; `&`/`*` bind to the name; function-definition braces on their own line; `m_` members, `k` constants; C-style casts with a space; `nullptr`; header guard `YT_JSON_JSON_H`; every file starts with a header comment saying what it is and why; no `TODO`/`FIXME`/commented-out code.
- Done = warning-free build and all ctests green in `build`, `build-asan`, `build-tsan`, `build-none`; `tst_json` also passes cross-built for ARMv7 under qemu (Task 1 Step 6).
- Commit only when the repository owner has said so for this plan (AGENTS.md §10). Commit steps show the message; skip `git commit` if commits were not authorized. ## Review Focus

1. **A body cut short by the network** (`{"a":[1,2`, an unterminated string, a lone `-`): `ok()` is false, every lookup is absent, nothing reads past the end — `rejectsMalformed` in Task 1, and the fuzzer in Task 3 under ASan.
2. **A crafted, deeply nested answer**: refused whole at depth 129, and `find()` on a legal 127-deep document must not recurse — `depthBound` and `findIsIterative` in Task 1.
3. **Keys and strings with escapes** (`"ab"` is the key `ab`, `\"` inside a key, surrogate pairs, a lone surrogate): lookups match by value, not by spelling — `escapedKeysMatchTheirValue`, `stringsDecodeOnceAndStayValid` in Task 1.
4. **A cached Doc read many times** (plan 2 caches replies): an escaped string decodes once and keeps the same view, so memory does not grow with reads — `stringsDecodeOnceAndStayValid` checks the pointer is stable.
5. **Control bytes in written strings** (a comment containing `\x01`): the writer emits `\u0001`, never the raw byte — Glaze wrote control bytes raw (invalid JSON); `writer` in Task 2.

```cpp
TEST_CASE("pathsAndAbsence") {
    Doc d(R"({"a":{"b":[1,{"c":"x"}]},"n":null,"t":true})");
    REQUIRE(d.ok());
    CHECK(d["a"]["b"][1]["c"].str() == "x");

    CHECK(d["a"]["b"][0]["c"].str().empty());

TEST_CASE("membersAndItemsInDocumentOrder") {
    Doc d(R"({"z":1,"a":2,"m":[3,4,5]})");
    std::string keys;
    for (Member m : d.root().members()) keys += m.key;
    CHECK(keys == "zam");
    int64_t sum = 0;
    for (Value v : d["m"].items()) sum += v.int64();
    CHECK(sum == 12);
    CHECK(d["m"].size() == 3);
    CHECK(d.root().size() == 3);
    for (Value v : d["z"].items()) FAIL("a number has no items " << v.raw());
}

TEST_CASE("duplicateKeysFirstWins") {
    Doc d(R"({"k":1,"k":2})");

    Doc d(R"({"plain":"abc","esc":"a\"b\\c\/d\n\u00e9\ud83d\ude00","lone":"\ud800x"})");
    REQUIRE(d.ok());
    CHECK(d["plain"].str() == "abc");
    CHECK(d["plain"].str().data() == d.text().data() + 10);   // a view into the body
    CHECK(d["esc"].str() == "a\"b\\c/d\n\xC3\xA9\xF0\x9F\x98\x80");
    CHECK(d["esc"].str().data() == d["esc"].str().data());     // decoded once, then reused
    CHECK(d["lone"].str() == "\xED\xA0\x80x");                  // WTF-8, as documented
}

    Doc d(R"({"\u0061b":1,"c\"d":2})");

    CHECK(d["c\"d"].int64() == 2);
    CHECK_FALSE(d["a"]);                          // a prefix of a key is not the key

    Doc d(R"([0,-12,9223372036854775807,9223372036854775808,1.5,2e3,-0.25])");

    REQUIRE(d.ok());

TEST_CASE("findWalksInDocumentOrder") {
    Doc d(R"({"r":{"x":1},"l":[{"r":{"x":2,"r":{"x":3}}},{"q":{"r":{"x":4}}}]})");
    REQUIRE(d.ok());
    std::vector<int64_t> seen;
    SUBCASE("Skip does not look inside a match") {
        d.root().find({"r"}, [&](std::string_view, Value v) { seen.push_back(v["x"].int64()); return Visit::Skip; });
        CHECK(seen == std::vector<int64_t>{1, 2, 4});
    }
    SUBCASE("Descend also visits matches inside a match") {
        d.root().find({"r"}, [&](std::string_view, Value v) { seen.push_back(v["x"].int64()); return Visit::Descend; });
        CHECK(seen == std::vector<int64_t>{1, 2, 3, 4});
    }
    SUBCASE("SkipRest leaves the rest of the holding object") {
        Doc e(R"({"o":{"r":{"x":1},"r2":{"r":{"x":9}}},"p":{"r":{"x":2}}})");
        e.root().find({"r"}, [&](std::string_view, Value v) { seen.push_back(v["x"].int64()); return Visit::SkipRest; });
        CHECK(seen == std::vector<int64_t>{1, 2});
    }
    SUBCASE("Stop ends the walk at the first match") {
        d.root().find({"r"}, [&](std::string_view, Value v) { seen.push_back(v["x"].int64()); return Visit::Stop; });
        CHECK(seen == std::vector<int64_t>{1});
    }
    SUBCASE("several keys, the matched key is reported") {
        std::string keys;
        d.root().find({"q", "x"}, [&](std::string_view k, Value) { keys += k; return Visit::Descend; });
        CHECK(keys == "xxxqx");
    }
}

TEST_CASE("findIsIterative") {
    std::string s;
    for (int i = 0; i < kMaxDepth - 1; ++i) s += "{\"k\":";
    s += "1";
    for (int i = 0; i < kMaxDepth - 1; ++i) s += "}";
    Doc d(s);
    REQUIRE(d.ok());
    int n = 0;
    d.root().find({"k"}, [&](std::string_view, Value) { ++n; return Visit::Descend; });
    CHECK(n == kMaxDepth - 1);
}

TEST_CASE("sharedBodyOutlivesTheCaller") {
    std::unique_ptr<Doc> d;
    {
        auto body = std::make_shared<const std::string>(R"({"a":"kept"})");
        d = std::make_unique<Doc>(body);
    }
    CHECK((*d)["a"].str() == "kept");
    CHECK_FALSE(Doc(std::shared_ptr<const std::string>()).ok());
}
```

```cpp
// yt::json — read JSON over an unmodified buffer, and write small JSON. The bottom of the
// layering: std only, no YouTube knowledge, C++17, so every parser and the transport read
// through it and it can leave the project as is.
//
// Reading is one validating pass that builds an index (the "tape"), then cheap views read
// lazily. A malloc per JSON node is what ruled out a DOM on the N9; re-skipping siblings
// at every level is what ruled out lazy views without an index (measured 13x slower than
// this). Design and measurements: docs/specs/2026-10-16-json-design.md.
//
// Tape: one u32 per token, two per object/array. Word = kind (2 bits) | flag (1 bit) |
// offset (29 bits). The flag marks a string/key that has escapes, or an object (vs array).
// An object/array's second word is the tape index just past its matching close, so any
// subtree is skipped in O(1). Strings are never copied unless they have escapes.
#ifndef YT_JSON_JSON_H
#define YT_JSON_JSON_H
#include <cstddef>
#include <cstdint>
#include <cstring>
#include <functional>
#include <initializer_list>
#include <memory>
#include <string>
#include <string_view>
#include <unordered_map>
#include <vector>

namespace yt::json {

// Nesting bound. Real InnerTube answers nest under 50 levels; a deeper document is refused
// whole, so a crafted answer cannot exhaust a 1 GB-RAM phone. Every walk is iterative.
inline constexpr int kMaxDepth = 128;
// Offsets are 29 bits: a larger document is refused.
inline constexpr size_t kMaxSize = (size_t) 1 << 29;

// What find() does after its callback saw a match.
enum class Visit {
    Descend,    // also look inside the matched value
    Skip,       // skip the matched value
    SkipRest,   // skip the matched value and the rest of the object that holds it
    Stop,       // end the walk (the first match is enough)
};

namespace detail {
enum : uint32_t {
    kScalar = 0u << 30,
    kString = 1u << 30,
    kKey = 2u << 30,
    kOpen = 3u << 30,
    kKindMask = 3u << 30,
    kFlag = 1u << 29,
    kOffMask = kFlag - 1,
};
}

class Doc;
class Items;
class Members;

// A value inside a Doc, or absent. Absence propagates: v["a"]["b"].str() is "" when any
// step is missing or has another type, so a path needs no checks between its steps.
// Two words, copied by value; valid while its Doc lives.
class Value {
public:
    Value() = default;
    explicit operator bool() const { return m_d != nullptr; }
    bool isObject() const;
    bool isArray() const;
    bool isString() const;
    bool isNumber() const;
    bool isBool() const;
    bool isNull() const;

    Value operator[](std::string_view key) const;   // the first member named key
    Value operator[](size_t index) const;           // O(index): iterate with items()
    template <class... K> Value path(K... keys) const
    {
        Value v = *this;
        ((v = v[keys]), ...);
        return v;
    }
    Value last() const;                              // the last array element
    size_t size() const;                             // elements or members; 0 otherwise

    std::string_view str() const;                    // a string's value; "" otherwise
    std::string string() const { return std::string(str()); }
    bool toInt(int64_t &out) const;                  // an integral number that fits
    int64_t int64(int64_t fallback = 0) const;
    bool toDouble(double &out) const;
    bool boolean(bool fallback = false) const;
    std::string_view raw() const;                    // the value's exact JSON text

    Items items() const;                             // array elements, in order
    Members members() const;                         // object members, in document order

    // Every member below this value whose key is one of `keys`, in document order; `f`
    // decides how the walk goes on. Iterative: depth cannot overflow the C stack.
    void find(std::initializer_list<std::string_view> keys,
              const std::function<Visit(std::string_view key, Value value)> &f) const;

private:
    friend class Doc;
    friend class Items;
    friend class Members;
    Value(const Doc *d, uint32_t i) : m_d(d), m_i(i) {}
    uint32_t word() const;
    char lead() const;
    const Doc *m_d = nullptr;
    uint32_t m_i = 0;
};

struct Member {
    std::string_view key;
    Value value;
};

class Items {
public:
    class iterator {
    public:
        Value operator*() const { return Value(m_d, m_i); }
        iterator &operator++();
        bool operator!=(const iterator &o) const { return m_i != o.m_i; }
    private:
        friend class Items;
        iterator(const Doc *d, uint32_t i) : m_d(d), m_i(i) {}
        const Doc *m_d;
        uint32_t m_i;
    };
    iterator begin() const { return iterator(m_d, m_begin); }
    iterator end() const { return iterator(m_d, m_end); }
private:
    friend class Value;
    Items(const Doc *d, uint32_t b, uint32_t e) : m_d(d), m_begin(b), m_end(e) {}
    const Doc *m_d;
    uint32_t m_begin, m_end;
};

class Members {
public:
    class iterator {
    public:
        Member operator*() const;
        iterator &operator++();
        bool operator!=(const iterator &o) const { return m_i != o.m_i; }
    private:
        friend class Members;
        iterator(const Doc *d, uint32_t i) : m_d(d), m_i(i) {}
        const Doc *m_d;
        uint32_t m_i;
    };
    iterator begin() const { return iterator(m_d, m_begin); }
    iterator end() const { return iterator(m_d, m_end); }
private:
    friend class Value;
    Members(const Doc *d, uint32_t b, uint32_t e) : m_d(d), m_begin(b), m_end(e) {}
    const Doc *m_d;
    uint32_t m_begin, m_end;
};

// An indexed JSON document. Validates RFC 8259 grammar, numbers, literals, escapes and raw
// control bytes, nesting <= kMaxDepth, size < kMaxSize; anything else leaves it !ok() and
// every lookup absent. UTF-8 is not validated: bytes pass through as sent, and a lone
// \uD800 escape decodes as WTF-8. Read from one thread at a time (escaped strings decode
// into the Doc on first read). Not copyable or movable: Values point at it.
class Doc {
public:
    explicit Doc(std::string_view json);                       // the caller keeps json alive
    explicit Doc(std::shared_ptr<const std::string> body);      // the Doc keeps body alive
    Doc(const Doc &) = delete;
    Doc &operator=(const Doc &) = delete;

    bool ok() const { return !m_tape.empty(); }
    Value root() const { return ok() ? Value(this, 0) : Value(); }
    Value operator[](std::string_view key) const { return root()[key]; }
    std::string_view text() const { return m_src; }
    size_t indexBytes() const { return m_tape.capacity() * sizeof(uint32_t); }

private:
    friend class Value;
    friend class Items;
    friend class Members;
    uint32_t off(uint32_t i) const { return m_tape[i] & detail::kOffMask; }
    uint32_t kind(uint32_t i) const { return m_tape[i] & detail::kKindMask; }
    uint32_t next(uint32_t i) const { return kind(i) == detail::kOpen ? m_tape[i + 1] : i + 1; }
    std::string_view strAt(uint32_t i) const;
    std::string_view decodeAt(uint32_t i) const;
    bool keyIs(uint32_t i, std::string_view k) const;
    void build();

    std::shared_ptr<const std::string> m_body;
    std::string_view m_src;
    std::vector<uint32_t> m_tape;
    mutable std::unordered_map<uint32_t, std::string> m_decoded;   // escaped strings, by index
};

// ---- inline accessors (the hot path of every parser) ----------------------------------------
inline uint32_t Value::word() const { return m_d->m_tape[m_i]; }
inline char Value::lead() const { return m_d->m_src[word() & detail::kOffMask]; }
inline bool Value::isObject() const
{
    return m_d && (word() & (detail::kKindMask | detail::kFlag)) == (detail::kOpen | detail::kFlag);
}
inline bool Value::isArray() const
{
    return m_d && (word() & (detail::kKindMask | detail::kFlag)) == detail::kOpen;
}
inline bool Value::isString() const { return m_d && (word() & detail::kKindMask) == detail::kString; }
inline bool Value::isNumber() const
{
    return m_d && (word() & detail::kKindMask) == detail::kScalar && lead() != 't' && lead() != 'f'
        && lead() != 'n';
}
inline bool Value::isBool() const
{
    return m_d && (word() & detail::kKindMask) == detail::kScalar && (lead() == 't' || lead() == 'f');
}
inline bool Value::isNull() const { return m_d && (word() & detail::kKindMask) == detail::kScalar && lead() == 'n'; }

inline bool Doc::keyIs(uint32_t i, std::string_view k) const
{
    const uint32_t w = m_tape[i];
    if (w & detail::kFlag) return decodeAt(i) == k;
    const size_t o = (w & detail::kOffMask) + 1;
    return o + k.size() < m_src.size() && m_src[o + k.size()] == '"'
        && std::memcmp(m_src.data() + o, k.data(), k.size()) == 0;
}
inline std::string_view Doc::strAt(uint32_t i) const
{
    const uint32_t w = m_tape[i];
    if (w & detail::kFlag) return decodeAt(i);
    const char *s = m_src.data() + (w & detail::kOffMask) + 1;
    const char *q = (const char *) std::memchr(s, '"', (size_t)(m_src.data() + m_src.size() - s));
    return std::string_view(s, (size_t)(q - s));
}

inline Value Value::operator[](std::string_view key) const
{
    if (!isObject()) return {};
    for (uint32_t i = m_i + 2, end = m_d->m_tape[m_i + 1]; i < end; i = m_d->next(i + 1))
        if (m_d->keyIs(i, key)) return Value(m_d, i + 1);
    return {};
}
inline Value Value::operator[](size_t index) const
{
    if (!isArray()) return {};
    for (uint32_t i = m_i + 2, end = m_d->m_tape[m_i + 1]; i < end; i = m_d->next(i))
        if (index-- == 0) return Value(m_d, i);
    return {};
}
inline Value Value::last() const
{
    if (!isArray()) return {};
```

```cpp
// yt::json — the index builder, the escape decoder, raw extents, find() and the writer.
// The builder is the only code that reads untrusted bytes structurally: one forward pass,
// no recursion, every byte read bounds-checked; the views trust the tape it produced.
#include "json/json.h"
#include <charconv>
#if defined(__SSE2__) || defined(_M_X64)
#include <emmintrin.h>
#define YT_JSON_SSE2 1
#elif defined(__ARM_NEON)
#include <arm_neon.h>
#define YT_JSON_NEON 1
#endif
#if defined(_MSC_VER)
#include <intrin.h>
#endif

namespace yt::json {

namespace {

using namespace detail;

// Bytes that end a number or literal: structure or whitespace.
struct Delimiters {
    bool t[256] = {};
    Delimiters() { for (unsigned char c : {',', '}', ']', ':', ' ', '\n', '\r', '\t'}) t[c] = true; }
};
const Delimiters kDelim;

// Index of the lowest set bit (x != 0).
int lowestBit(uint64_t x)
{
#if defined(_MSC_VER)
    unsigned long i;
    _BitScanForward64(&i, x);
    return (int) i;
#else
    return __builtin_ctzll((unsigned long long) x);
#endif
}

bool isBlank(char c) { return c == ' ' || c == '\n' || c == '\r' || c == '\t'; }
bool isDigit(char c) { return c >= '0' && c <= '9'; }
bool isHex(char c) { return isDigit(c) || ((c | 0x20) >= 'a' && (c | 0x20) <= 'f'); }

// The first '"', '\\' or control byte (invalid raw inside a string) at or after p; e if none.
// 83-91% of an InnerTube answer is string bytes: this loop is the parser's speed. SSE2 and
// NEON take 16 bytes a step; elsewhere one machine word (SWAR).
const char *stringStop(const char *p, const char *e)
{
#if defined(YT_JSON_SSE2)
    const __m128i quote = _mm_set1_epi8('"'), slash = _mm_set1_epi8('\\'), ctl = _mm_set1_epi8(0x1F);
    while (e - p >= 16) {
        const __m128i v = _mm_loadu_si128((const __m128i *) p);
        const __m128i low = _mm_cmpeq_epi8(_mm_min_epu8(v, ctl), v);          // v <= 0x1F
        const unsigned m = (unsigned) _mm_movemask_epi8(
            _mm_or_si128(_mm_or_si128(_mm_cmpeq_epi8(v, quote), _mm_cmpeq_epi8(v, slash)), low));
        if (m) return p + lowestBit(m);
        p += 16;
    }
#elif defined(YT_JSON_NEON)
    const uint8x16_t quote = vdupq_n_u8('"'), slash = vdupq_n_u8('\\'), space = vdupq_n_u8(0x20);
    while (e - p >= 16) {
        const uint8x16_t v = vld1q_u8((const uint8_t *) p);
        const uint8x16_t m = vorrq_u8(vorrq_u8(vceqq_u8(v, quote), vceqq_u8(v, slash)), vcltq_u8(v, space));
        const uint8x8_t nib = vshrn_n_u16(vreinterpretq_u16_u8(m), 4);       // 4 bits per byte
        uint64_t bits;
        std::memcpy(&bits, &nib, 8);
        if (bits) return p + (lowestBit(bits) >> 2);
        p += 16;
    }
#else
    using W = uintptr_t;
    const W ones = (W) 0x0101010101010101ull, highs = ones << 7;
    while ((size_t)(e - p) >= sizeof(W)) {
        W w;
        std::memcpy(&w, p, sizeof w);
        const W a = w ^ (ones * '"'), b = w ^ (ones * '\\');
        // A zero byte of a or b, or a byte below 0x20, sets its high bit; the lowest flag is
        // exact (borrows only run upwards), so on little-endian it is the answer.
        const W z = (((a - ones) & ~a) | ((b - ones) & ~b) | (w - ones * 0x20)) & ~w & highs;
        if (z) {
#if defined(__BYTE_ORDER__) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
            return p + lowestBit((uint64_t) z) / 8;
#else
            break;
#endif
        }
        p += sizeof(W);
    }
#endif
    while (p < e && *p != '"' && *p != '\\' && (unsigned char) *p >= 0x20) ++p;
    return p;
}

// An RFC 8259 §6 number or a literal at p, followed by a delimiter or the end: its end, or
// nullptr when malformed.
const char *scalarEnd(const char *p, const char *e)
{
    auto literal = [&](const char *w, size_t n) -> const char * {
        return (size_t)(e - p) >= n && std::memcmp(p, w, n) == 0 ? p + n : nullptr;
    };
    const char *q;
    if (*p == 't') q = literal("true", 4);
    else if (*p == 'f') q = literal("false", 5);
    else if (*p == 'n') q = literal("null", 4);
    else {
        q = p;
        if (q < e && *q == '-') ++q;
        if (q < e && *q == '0') ++q;
        else if (q < e && *q >= '1' && *q <= '9') while (q < e && isDigit(*q)) ++q;
        else return nullptr;
        if (q < e && *q == '.') {
            if (++q >= e || !isDigit(*q)) return nullptr;
            while (q < e && isDigit(*q)) ++q;
        }
        if (q < e && (*q == 'e' || *q == 'E')) {
            ++q;
            if (q < e && (*q == '+' || *q == '-')) ++q;
            if (q >= e || !isDigit(*q)) return nullptr;
            while (q < e && isDigit(*q)) ++q;
        }
    }
    if (!q || (q < e && !kDelim.t[(unsigned char) *q])) return nullptr;
    return q;
}

// Past the closing quote of the string whose opening quote is at s (already validated).
const char *stringEnd(const char *s)
{
    const char *q = s + 1;
    while (*q != '"') q += *q == '\\' ? (q[1] == 'u' ? 6 : 2) : 1;
    return q + 1;
}

void Doc::build()
{
    const char *const b = m_src.data(), *p = b, *const e = b + m_src.size();
    if (m_src.size() >= kMaxSize) return;
    // ~15 bytes per token in InnerTube answers: a tenth of the size rarely grows.
    m_tape.resize(m_src.size() / 10 + 16);
    uint32_t *t = m_tape.data();
    size_t cap = m_tape.size(), n = 0;
    uint32_t openAt[kMaxDepth];
    bool inObject[kMaxDepth];
    int depth = 0;
    // What the grammar allows next.
    enum : uint8_t { Val, KeyOrEnd, Key, Colon, CommaOrEnd, ValOrEnd, Done } st = Val;
    auto fail = [&] { m_tape.clear(); m_tape.shrink_to_fit(); };
    for (;;) {
        if (p >= e) break;
        if ((unsigned char) *p <= ' ') {
            while (p < e && isBlank(*p)) ++p;
            if (p >= e) break;
        }
        if (cap - n < 2) {
            m_tape.resize(cap += cap / 2);
            t = m_tape.data();
        }
        const char c = *p;
        const uint32_t o = (uint32_t)(p - b);
        switch (c) {
        case '"': {
            uint32_t kind;
            if (st == KeyOrEnd || st == Key) { kind = kKey; st = Colon; }
            else if (st == Val || st == ValOrEnd) { kind = kString; st = CommaOrEnd; }
            else return fail();
            const char *q = p + 1;
            for (;;) {
                q = stringStop(q, e);
                if (q >= e) return fail();
                if (*q == '"') break;
                if (*q != '\\' || e - q < 2) return fail();   // a raw control byte, or cut short
                switch (q[1]) {
                case '"': case '\\': case '/': case 'b': case 'f': case 'n': case 'r': case 't':
                    q += 2;
                    break;
                case 'u':
                    if (e - q < 6 || !isHex(q[2]) || !isHex(q[3]) || !isHex(q[4]) || !isHex(q[5]))
                        return fail();
                    q += 6;
                    break;
                default:
                    return fail();
                }
                kind |= kFlag;
            }
            t[n++] = kind | o;
            p = q + 1;
            break;
        }
        case ':':
            if (st != Colon) return fail();
            ++p;
            st = Val;
            break;
        case ',':
            if (st != CommaOrEnd) return fail();
            ++p;
            st = inObject[depth - 1] ? Key : Val;
            break;
        case '{': case '[':
            if (st != Val && st != ValOrEnd) return fail();
            if (depth == kMaxDepth) return fail();
            inObject[depth] = c == '{';
            openAt[depth++] = (uint32_t) n;
            t[n++] = kOpen | (c == '{' ? (uint32_t) kFlag : 0u) | o;
            t[n++] = 0;
            ++p;
            st = c == '{' ? KeyOrEnd : ValOrEnd;
            break;
        case '}': case ']':
            if (depth == 0 || inObject[depth - 1] != (c == '}')) return fail();
            if (st != CommaOrEnd && st != (c == '}' ? KeyOrEnd : ValOrEnd)) return fail();
            t[openAt[--depth] + 1] = (uint32_t) n;
            ++p;
            st = CommaOrEnd;
            break;
        default:
            if (st != Val && st != ValOrEnd) return fail();
            t[n++] = kScalar | o;
            if (!(p = scalarEnd(p, e))) return fail();
            st = CommaOrEnd;
            break;
        }
        if (depth == 0 && st == CommaOrEnd) {
            st = Done;
            break;
        }
    }
    while (p < e && isBlank(*p)) ++p;
    if (st != Done || p != e) return fail();
    m_tape.resize(n);
}

std::string_view Doc::decodeAt(uint32_t i) const
{
    if (auto it = m_decoded.find(i); it != m_decoded.end()) return it->second;
    auto hex = [](const char *h) {
        uint32_t v = 0;
        for (int k = 0; k < 4; ++k) v = (v << 4) | (uint32_t)(isDigit(h[k]) ? h[k] - '0' : (h[k] | 0x20) - 'a' + 10);
        return v;
    };
    std::string out;
    auto utf8 = [&](uint32_t cp) {
        if (cp < 0x80) {
            out += (char) cp;
            return;
        }
        if (cp < 0x800) out += (char)(0xC0 | (cp >> 6));
        else {
            if (cp < 0x10000) out += (char)(0xE0 | (cp >> 12));
            else {
                out += (char)(0xF0 | (cp >> 18));
                out += (char)(0x80 | ((cp >> 12) & 0x3F));
            }
            out += (char)(0x80 | ((cp >> 6) & 0x3F));
        }
        out += (char)(0x80 | (cp & 0x3F));
    };
    // The builder validated every escape: this loop trusts the syntax.
    for (const char *p = m_src.data() + off(i) + 1; *p != '"';) {
        if (*p != '\\') {
            out += *p++;
            continue;
        }
        const char x = p[1];
        p += 2;
        switch (x) {
        case 'n': out += '\n'; break;
        case 't': out += '\t'; break;
        case 'r': out += '\r'; break;
        case 'b': out += '\b'; break;
        case 'f': out += '\f'; break;
        case 'u': {
            uint32_t cp = hex(p);
            p += 4;
            if (cp >= 0xD800 && cp < 0xDC00 && p[0] == '\\' && p[1] == 'u') {
                const uint32_t lo = hex(p + 2);
                if (lo >= 0xDC00 && lo < 0xE000) {
                    cp = 0x10000 + ((cp - 0xD800) << 10) + (lo - 0xDC00);
                    p += 6;
                }
            }
            utf8(cp);                                    // a lone surrogate stays (WTF-8)
            break;
        }
        default: out += x;                               // \" \\ \/
        }
    }
    return m_decoded.emplace(i, std::move(out)).first->second;
}

std::string_view Value::raw() const
{
    if (!m_d) return {};
    const Doc &d = *m_d;
    const char *const src = d.m_src.data(), *const e = src + d.m_src.size();
    // The tape keeps no end offsets: go down the chain of last children to a leaf (or an
    // empty container), take its end, then climb back over one close bracket per level.
    // (Using the next token's offset instead returned the outer bracket for "[[]   ]".)
    uint32_t cur = m_i;
    int levels = 0;
    const char *p;
    for (;;) {
        const uint32_t w = d.m_tape[cur];
        const char *s = src + (w & kOffMask);
        if ((w & kKindMask) == kScalar) {
            p = scalarEnd(s, e);
            break;
        }
        if ((w & kKindMask) != kOpen) {
            p = stringEnd(s);
            break;
        }
        ++levels;
        const bool obj = (w & kFlag) != 0;
        uint32_t lastChild = 0;
        for (uint32_t i = cur + 2, end = d.m_tape[cur + 1]; i < end; i = d.next(obj ? i + 1 : i))
            lastChild = obj ? i + 1 : i;
        if (!lastChild) {
            p = s + 1;
            break;
        }
        cur = lastChild;
    }
    for (; levels > 0; --levels) {
        while (isBlank(*p)) ++p;
        ++p;                                             // the validated close bracket
    }
    const char *s = src + (word() & kOffMask);
    return std::string_view(s, (size_t)(p - s));
}

void Value::find(std::initializer_list<std::string_view> keys,
                 const std::function<Visit(std::string_view, Value)> &f) const
{
    if (!m_d || (word() & kKindMask) != kOpen) return;
    const uint32_t *t = m_d->m_tape.data();
    const char *src = m_d->m_src.data();
    // First-byte filter: most keys are rejected without a comparison.
    uint32_t first[8] = {};
    for (std::string_view k : keys) {
        const unsigned c = k.empty() ? '"' : (unsigned char) k[0];
        first[c >> 5] |= 1u << (c & 31);
    }
    // ends[]: the tape end of every container being walked, innermost last.
    uint32_t ends[kMaxDepth + 1];
    int sp = 0;
    ends[sp++] = t[m_i + 1];
    uint32_t i = m_i + 2;
    while (sp > 0) {
        if (i >= ends[sp - 1]) {
            --sp;
            continue;
        }
        const uint32_t w = t[i];
        const uint32_t kind = w & kKindMask;
        if (kind == kOpen) {
            ends[sp++] = t[i + 1];
            i += 2;
            continue;
        }
        if (kind != kKey) {
            ++i;
            continue;
        }
        const unsigned c = (unsigned char) src[(w & kOffMask) + 1];
        bool hit = false;
        std::string_view key;
        if ((first[c >> 5] & (1u << (c & 31))) || (w & kFlag))
            for (std::string_view k : keys)
                if (m_d->keyIs(i, k)) {
                    hit = true;
                    key = k;
                    break;
                }
        if (!hit) {
            ++i;
            continue;
        }
        switch (f(key, Value(m_d, i + 1))) {
        case Visit::Descend: ++i; break;
        case Visit::Skip: i = m_d->next(i + 1); break;
        case Visit::SkipRest: i = ends[sp - 1]; break;
        case Visit::Stop: return;
        }
    }
}
```

```cpp
// Compact JSON output (no spaces), for request bodies and saved state. Strings are escaped
// per RFC 8259 §7 (\" \\ \b \f \n \r \t, other bytes below 0x20 as \u00XX); everything else,
// UTF-8 included, is written as is. Commas are placed automatically.
class Writer {
public:
    Writer &beginObject() { return open('{'); }
    Writer &endObject() { return close('}'); }
    Writer &beginArray() { return open('['); }
    Writer &endArray() { return close(']'); }
    Writer &key(std::string_view k);
    Writer &str(std::string_view s);
    Writer &num(int64_t n);
    Writer &boolean(bool b);
    Writer &raw(std::string_view json);              // a value that is already JSON text
    // Shorthands for one member.
    Writer &member(std::string_view k, std::string_view s) { return key(k).str(s); }
    Writer &member(std::string_view k, const char *s) { return key(k).str(s); }
    Writer &member(std::string_view k, int64_t n) { return key(k).num(n); }
    Writer &member(std::string_view k, int n) { return key(k).num(n); }
```

```cpp
    comma();

    for (char ch : s) {
        const unsigned char u = (unsigned char) ch;
        switch (ch) {
        case '"': m_out += "\\\""; break;
        case '\\': m_out += "\\\\"; break;
        case '\b': m_out += "\\b"; break;
        case '\f': m_out += "\\f"; break;
        case '\n': m_out += "\\n"; break;
        case '\r': m_out += "\\r"; break;
        case '\t': m_out += "\\t"; break;
        default:

                m_out += "\\u00";
                m_out += kHex[u >> 4];

            }
        }

    m_comma = true;
    return *this;
```

```cpp
// libFuzzer target for yt::json (INNERTUBE_FUZZ=ON, Clang). No second parser as oracle:
// the document is its own. For any input the builder must not crash; for an accepted one,
// writing the tree back must reparse to the same text (a round trip through Writer),
// every value's raw() must reparse, and find() must visit every member exactly once.
#include "json/json.h"
#include <cstdlib>
#include <string>
using namespace yt::json;

namespace {
void check(bool c) { if (!c) std::abort(); }

// Canonical text of v: members/items through the views, numbers and literals as raw text.
void write(Writer &w, Value v, size_t &members)
{
    if (v.isObject()) {
        w.beginObject();
        for (Member m : v.members()) {
            ++members;
            w.key(m.key);
            write(w, m.value, members);
        }
        w.endObject();
    } else if (v.isArray()) {
        w.beginArray();
        size_t n = 0;
        for (Value x : v.items()) {
            write(w, x, members);
            ++n;
        }
        check(n == v.size());
        w.endArray();
    } else if (v.isString()) {
        w.str(v.str());
    } else {
        w.raw(v.raw());
    }
    Doc again(v.raw());
    check(again.ok());
}
} // namespace

extern "C" int LLVMFuzzerTestOneInput(const unsigned char *data, size_t size)
{
    Doc d(std::string_view((const char *) data, size));
    if (!d.ok()) return 0;
    Writer w;
    size_t members = 0;
    write(w, d.root(), members);
    Doc back(w.text());
    check(back.ok());
    Writer w2;
    size_t members2 = 0;
    write(w2, back.root(), members2);
    check(w2.text() == w.text() && members2 == members);
    // find() with every key that occurs: the callback sees each member once.
    size_t found = 0;
    std::string keys[4];
    int nk = 0;
    for (Member m : d.root().members())
        if (nk < 4) keys[nk++] = std::string(m.key);
    if (nk > 0) {
        size_t expect = 0;
        d.root().find({keys[0], keys[1 % nk], keys[2 % nk], keys[3 % nk]}, [&](std::string_view, Value) {
            ++found;
            return Visit::Descend;
        });
        // Count the same matches without find(): a naive recursive walk.
        auto count = [&](auto &&self, Value v) -> void {
            for (Member m : v.members()) {
                for (int i = 0; i < nk; ++i)
                    if (m.key == keys[i]) { ++expect; break; }
                self(self, m.value);
            }
            for (Value x : v.items()) self(self, x);
        };
        count(count, d.root());
        check(found == expect);
    }
    return 0;
}
```
